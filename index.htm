                    <span class="text-xs sm:text-sm leading-relaxed">${opt}</span>

                  </label>

                `;

              }).join('')}

            </div>

          `;

        }

        // TYPE 2: Multiple Choice Multiple Answers (MCMA)

        else if (q.type === 'mcma') {

          const selectedSet = userAnswers[q.id] || [];

          inputControlsHtml = `

            <div class="space-y-2">

              ${q.options.map((opt, optIdx) => {

                const isChecked = selectedSet.includes(optIdx);

                return `

                  }">

                    <input type="checkbox"

                           name="${q.id}"

                           value="${optIdx}"

                           ${isChecked ? 'checked' : ''}

                           onchange="handleMcmaToggle('${q.id}', ${optIdx})"

                           class="mt-0.5 w-4 h-4 rounded text-lavender-600 focus:ring-lavender-400 border-slate-300">

                    <span class="text-xs sm:text-sm leading-relaxed">${opt}</span>

                  </label>

                `;

              }).join('')}

            </div>

            <p class="text-[11px] text-slate-500 mt-2 italic flex items-center gap-1">

              <i data-lucide="info" class="w-3.5 h-3.5 text-lavender-500"></i> All required correct choices must be selected for credit.

            </p>

          `;

        }

        // TYPE 3: Category Matching Question

        else if (q.type === 'category') {

          const catSelections = userAnswers[q.id] || {};

          inputControlsHtml = `

            <div class="space-y-3">

              ${q.items.map((item, itemIdx) => {

                const currentChoice = catSelections[itemIdx] || "";

                return `

                  <div class="p-3 rounded-xl bg-white border border-slate-200 space-y-2">

                    <p class="text-xs sm:text-sm font-medium text-slate-800 leading-snug">

                    </p>

                    <div class="flex flex-wrap items-center gap-2 pt-1">

                      <span class="text-[11px] font-semibold text-slate-500">Assign Category:</span>

                      ${q.categories.map(cat => {

                        const isCatSelected = currentChoice === cat;

                        return `

                          <button type="button"

                                  onclick="handleCategorySelect('${q.id}', ${itemIdx}, '${cat}')"

                                  class="px-2.5 py-1 rounded-lg text-xs font-medium border transition-all ${

                                    isCatSelected

                                      ? 'bg-lavender-600 text-white border-lavender-600 shadow-2xs'

                                      : 'bg-slate-50 hover:bg-lavender-50 text-slate-600 border-slate-200'

                                  }">

                            ${cat}

                          </button>

                        `;

                      }).join('')}

                    </div>

                  </div>

                `;

              }).join('')}

            </div>

          `;

        }

 

        // Feedback / Explanation Box (rendered if evaluated)

        let explanationHtml = '';

        if (isEvaluated) {

          const evaluationResult = evaluateSingleQuestion(q);

          const isCorrect = evaluationResult.isCorrect;

          explanationHtml = `

            <div class="mt-3.5 p-3 rounded-xl border text-xs leading-relaxed ${

              isCorrect ? 'bg-emerald-50 border-emerald-300 text-emerald-900' : 'bg-rose-50 border-rose-300 text-rose-900'

            }">

              <div class="flex items-center gap-1.5 font-bold mb-1">

                <i data-lucide="${isCorrect ? 'check-circle-2' : 'x-circle'}" class="w-4 h-4 ${isCorrect ? 'text-emerald-600' : 'text-rose-600'}"></i>

                <span>${isCorrect ? 'Correct! (+1.0 point)' : 'Incorrect / Incomplete (0 points)'}</span>

              </div>

              <p class="mt-0.5">${q.explanation}</p>

            </div>

          `;

        }

 

        qCard.innerHTML = headerHtml + inputControlsHtml + explanationHtml;

        container.appendChild(qCard);

      });

 

      lucide.createIcons();

    }

 

    /* ========================================================================

       INTERACTION HANDLERS

       ======================================================================== */

    function handleMcqSelect(qId, optionIndex) {

      if (isEvaluated) isEvaluated = false;

      userAnswers[qId] = optionIndex;

      renderQuestions();

      updateProgress();

    }

 

    function handleMcmaToggle(qId, optionIndex) {

      if (isEvaluated) isEvaluated = false;

      if (!userAnswers[qId]) userAnswers[qId] = [];

      const set = userAnswers[qId];

      const pos = set.indexOf(optionIndex);

      if (pos > -1) {

        set.splice(pos, 1);

      } else {

        set.push(optionIndex);

      }

      userAnswers[qId] = [...set];

      renderQuestions();

      updateProgress();

    }

 

    function handleCategorySelect(qId, itemIndex, categoryName) {

      if (isEvaluated) isEvaluated = false;

      if (!userAnswers[qId]) userAnswers[qId] = {};

      userAnswers[qId][itemIndex] = categoryName;

      renderQuestions();

      updateProgress();

    }

 

    /* ========================================================================

       PROGRESS TRACKER

       ======================================================================== */

    function updateProgress() {

      const story = STORIES_DATABASE[currentLevel][currentStoryIndex];

      let answeredCount = 0;

 

      story.questions.forEach(q => {

        if (q.type === 'mcq') {

          if (userAnswers[q.id] !== undefined) answeredCount++;

        } else if (q.type === 'mcma') {

          if (userAnswers[q.id] && userAnswers[q.id].length > 0) answeredCount++;

        } else if (q.type === 'category') {

          const ans = userAnswers[q.id];

          if (ans && Object.keys(ans).length === q.items.length) {

            answeredCount++;

          }

        }

      });

 

      const total = story.questions.length;

      document.getElementById('answeredCountBadge').textContent = `${answeredCount} / ${total} Answered`;

      const pct = Math.round((answeredCount / total) * 100);

      document.getElementById('progressBar').style.width = `${pct}%`;

    }

 

    /* ========================================================================

       EVALUATION LOGIC

       ======================================================================== */

    function evaluateSingleQuestion(q) {

      const userAns = userAnswers[q.id];

 

      if (q.type === 'mcq') {

        const isCorrect = userAns === q.correct;

        return { isCorrect, score: isCorrect ? 1 : 0 };

      }

      else if (q.type === 'mcma') {

        if (!userAns || !Array.isArray(userAns)) return { isCorrect: false, score: 0 };

        const sortedUser = [...userAns].sort((a,b) => a-b);

        const sortedCorrect = [...q.correct].sort((a,b) => a-b);

        const isExactMatch = sortedUser.length === sortedCorrect.length &&

                             sortedUser.every((val, index) => val === sortedCorrect[index]);

        return { isCorrect: isExactMatch, score: isExactMatch ? 1 : 0 };

      }

      else if (q.type === 'category') {

        if (!userAns) return { isCorrect: false, score: 0 };

        let allCorrect = true;

        q.items.forEach((item, idx) => {

          if (userAns[idx] !== item.correctCategory) {

            allCorrect = false;

          }

        });

        return { isCorrect: allCorrect, score: allCorrect ? 1 : 0 };

      }

 

      return { isCorrect: false, score: 0 };

    }

 

    function evaluateQuiz() {

      const story = STORIES_DATABASE[currentLevel][currentStoryIndex];

     

      let unAnsweredIndex = -1;

      for (let i = 0; i < story.questions.length; i++) {

        const q = story.questions[i];

        if (q.type === 'mcq' && userAnswers[q.id] === undefined) {

          unAnsweredIndex = i; break;

        }

        if (q.type === 'mcma' && (!userAnswers[q.id] || userAnswers[q.id].length === 0)) {

          unAnsweredIndex = i; break;

        }

        if (q.type === 'category') {

          const ans = userAnswers[q.id];

          if (!ans || Object.keys(ans).length < q.items.length) {

            unAnsweredIndex = i; break;

          }

        }

      }

 

      if (unAnsweredIndex !== -1) {

        showToast(`Please answer Question ${unAnsweredIndex + 1} before checking!`, 'warning');

        const targetCard = document.getElementById(`qCard_${story.questions[unAnsweredIndex].id}`);

        if (targetCard) targetCard.scrollIntoView({ behavior: 'smooth', block: 'center' });

        return;

      }

 

      isEvaluated = true;

      let totalScore = 0;

      story.questions.forEach(q => {

        const res = evaluateSingleQuestion(q);

        totalScore += res.score;

      });

 

      renderQuestions();

      showFeedbackPanel(totalScore, story.questions.length);

      showToast(`Assessment complete! Score: ${totalScore}/5`, 'success');

      lucide.createIcons();

    }

 

    function resetCurrentQuiz() {

      stopAudio();

      userAnswers = {};

      isEvaluated = false;

      renderQuestions();

      updateProgress();

      hideFeedbackPanel();

      showToast("Quiz reset. Ready for a fresh attempt!", 'info');

      lucide.createIcons();

    }

    function readAloudStory() {

      if (!('speechSynthesis' in window)) {

        showToast("Text-to-speech is not supported on this browser.", "warning");

        return;

      }

 

      const label = document.getElementById('readAloudLabel');

 

      if (isSpeaking) {

        stopAudio();

        return;

      }

 

      const story = STORIES_DATABASE[currentLevel][currentStoryIndex];

      const fullText = story.title + ". " + story.paragraphs.join(" ");

 

      utterance = new SpeechSynthesisUtterance(fullText);

      utterance.lang = 'en-US';

      utterance.rate = 0.9;

      utterance.pitch = 1.0;

 

      utterance.onstart = () => {

        isSpeaking = true;

        label.textContent = "Pause Voice";

        showToast("Narrating story...", "info");

      };

 

      utterance.onend = () => {

        stopAudio();

      };

 

      utterance.onerror = () => {

        stopAudio();

      };

 

      synth.speak(utterance);

    }

 

    function stopAudio() {

      if (synth && synth.speaking) {

        synth.cancel();

      }

      isSpeaking = false;

      const label = document.getElementById('readAloudLabel');

      if (label) label.textContent = "Listen Story";

    }
